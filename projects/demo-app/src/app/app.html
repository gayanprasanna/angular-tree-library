💻 How to Use This Library

Getting started is super easy! Here's a quick example:

// 1. Install the library
npm install @gayanprasanna/ngx-tree

// 2. Import it in your component
import {{ '{' }} NgxTreeComponent, TreeNode, TreeConfig {{ '}' }} from '@gayanprasanna/ngx-tree';

@Component({{ '{' }}
  selector: 'my-awesome-component',
  imports: [NgxTreeComponent],
  template: `
    <ngx-tree
      [nodes]="myTreeData"
      [config]="myConfig"
      (nodeClick)="onNodeClick($event)"
      (nodeSelect)="onNodeSelect($event)"
      (lazyLoad)="onLazyLoad($event)"
    ></ngx-tree>
  `
{{ '}' }})
export class MyComponent {{ '{' }}
  // Your tree data
  myTreeData: TreeNode[] = [
    {{ '{' }}
      id: '1',
      label: 'My Projects',
      icon: '🚀',
      hasChildren: true,
      children: [
        {{ '{' }} id: '1-1', label: 'Awesome Project', icon: '⭐' {{ '}' }},
        {{ '{' }} id: '1-2', label: 'Another Project', icon: '💎' {{ '}' }}
      ]
    {{ '}' }}
  ];

  // Configuration
  myConfig: TreeConfig = {{ '{' }}
    showCheckboxes: true,
    showIcons: true,
    allowMultipleSelection: true,
    searchable: true,
    lazyLoad: true,
    
    // Control panel options
    showControls: true,
    showExpandAll: true,
    showCollapseAll: true,
    showClearSelection: true
  {{ '}' }};

  // Event handlers
  onNodeClick(event: TreeNodeClickEvent) {{ '{' }}
    console.log('Clicked:', event.node.label);
  {{ '}' }}

  onLazyLoad(event: LazyLoadEvent) {{ '{' }}
    // Load more data from your API
    this.myApiService.getChildren(event.node.id).subscribe(children => {{ '{' }}
      event.callback(children);
    {{ '}' }});
  {{ '}' }}

  // Programmatic control methods
  expandAll() {{ '{' }}
    this.treeComponent.expandAllProgrammatically();
  {{ '}' }}

  selectSpecificNode() {{ '{' }}
    this.treeComponent.selectNodeById('1-1');
  {{ '}' }}
{{ '}' }}
